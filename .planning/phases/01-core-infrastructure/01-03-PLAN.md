    **Database Connection** (`packages/bot/src/db/index.ts`):

    Per RESEARCH.md Pattern 5. Connect to SQLite with WAL mode:
    ```typescript
    import Database from 'better-sqlite3';
    import { drizzle } from 'drizzle-orm/better-sqlite3';
    import { migrate } from 'drizzle-orm/better-sqlite3/migrator';
    import * as schema from './schema.js';
    import type { Logger } from 'pino';
    import { mkdirSync } from 'node:fs';
    import { dirname } from 'node:path';

    export function initDatabase(dbPath: string, migrationsFolder: string, log: Logger) {
      // Ensure data directory exists
      mkdirSync(dirname(dbPath), { recursive: true });

      const sqlite = new Database(dbPath);
      // Enable WAL mode per RESEARCH.md Pitfall 4 + Pattern 5
      sqlite.pragma('journal_mode = WAL');

      const db = drizzle(sqlite, { schema });

      // Run migrations on startup per D-10
      log.info({ dbPath, migrationsFolder }, 'Running database migrations');
      migrate(db, { migrationsFolder });

      log.info({ dbPath }, 'Database initialized');
      return { db, sqlite };
    }
    ```

    **Game State Manager** (`packages/bot/src/state/game-state.ts`):

    Per D-09. Central state store that periodically refreshes from ogamed and persists to SQLite:
    ```typescript
    import { eq } from 'drizzle-orm';
    import type { BetterSQLite3Database } from 'drizzle-orm/better-sqlite3';
    import * as schema from '../db/schema.js';
    import type { OgamedClient } from '../client/ogamed-client.js';
    import type { Logger } from 'pino';

    export class GameState {
      private db: BetterSQLite3Database<typeof schema>;
      private client: OgamedClient;
      private log: Logger;
      private refreshIntervalMs: number;
      private refreshTimer: ReturnType<typeof setInterval> | null = null;

      constructor(
        db: BetterSQLite3Database<typeof schema>,
        client: OgamedClient,
        refreshIntervalMs: number,
        log: Logger,
      ) {
        this.db = db;
        this.client = client;
        this.refreshIntervalMs = refreshIntervalMs;
        this.log = log.child({ component: 'game-state' });
      }

      async start(): Promise<void> {
        this.log.info({ refreshIntervalMs: this.refreshIntervalMs }, 'Starting game state manager');
        // Initial refresh
        await this.refresh();
        // Periodic refresh with jitter (anti-detection: ±20% random variation)
        const scheduleNext = () => {
          const jitter = this.refreshIntervalMs * 0.2 * (Math.random() * 2 - 1);
          const delay = this.refreshIntervalMs + jitter;
          this.refreshTimer = setTimeout(async () => {
            try {
              await this.refresh();
            } catch (error) {
              this.log.error({ error: String(error) }, 'State refresh failed');
            }
            scheduleNext();
          }, delay);
        };
        scheduleNext();
      }

      stop(): void {
        if (this.refreshTimer) {
          clearTimeout(this.refreshTimer);
          this.refreshTimer = null;
        }
        this.log.info('Game state manager stopped');
      }

      async refresh(): Promise<void> {
        this.log.debug('Refreshing game state from ogamed');

        try {
          // Fetch all state in sequence (through rate limiter — one call at a time)
          const planets = await this.client.getPlanets();
          const fleets = await this.client.getFleets();
          const research = await this.client.getResearch();

          // Persist planets
          for (const planet of planets) {
            await this.db.insert(schema.planets)
              .values({
                id: planet.ID,
                name: planet.Name,
                galaxy: planet.Coordinate.galaxy,
                system: planet.Coordinate.system,
                position: planet.Coordinate.position,
                type: planet.Coordinate.type,
                diameter: planet.Diameter,
                fieldsUsed: planet.FieldsUsed,
                fieldsTotal: planet.FieldsTotal,
                tempMin: planet.TemperatureMin,
                tempMax: planet.TemperatureMax,
                updatedAt: new Date(),
              })
              .onConflictDoUpdate({
                target: schema.planets.id,
                set: {
                  name: planet.Name,
                  galaxy: planet.Coordinate.galaxy,
                  system: planet.Coordinate.system,
                  position: planet.Coordinate.position,
                  type: planet.Coordinate.type,
                  diameter: planet.Diameter,
                  fieldsUsed: planet.FieldsUsed,
                  fieldsTotal: planet.FieldsTotal,
                  tempMin: planet.TemperatureMin,
                  tempMax: planet.TemperatureMax,
                  updatedAt: new Date(),
                },
              });

            // Fetch and persist per-planet data
            try {
              const resources = await this.client.getResources(planet.ID);
              await this.db.insert(schema.resources)
                .values({
                  planetId: planet.ID,
                  metal: resources.Metal,
                  crystal: resources.Crystal,
                  deuterium: resources.Deuterium,
                  energy: resources.Energy,
                  updatedAt: new Date(),
                })
                .onConflictDoUpdate({
                  target: schema.resources.planetId,
                  set: {
                    metal: resources.Metal,
                    crystal: resources.Crystal,
                    deuterium: resources.Deuterium,
                    energy: resources.Energy,
                    updatedAt: new Date(),
                  },
                });
            } catch (error) {
              this.log.warn({ planetId: planet.ID, error: String(error) }, 'Failed to refresh resources');
            }

            try {
              const buildings = await this.client.getResourceBuildings(planet.ID);
              await this.db.insert(schema.resourceBuildings)
                .values({
                  planetId: planet.ID,
                  metalMine: buildings.MetalMine,
                  crystalMine: buildings.CrystalMine,
                  deuteriumSynthesizer: buildings.DeuteriumSynthesizer,
                  solarPlant: buildings.SolarPlant,
                  fusionReactor: buildings.FusionReactor,
                  metalStorage: buildings.MetalStorage,
                  crystalStorage: buildings.CrystalStorage,
                  deuteriumStorage: buildings.DeuteriumStorage,
                  updatedAt: new Date(),
                })
                .onConflictDoUpdate({
                  target: schema.resourceBuildings.planetId,
                  set: {
                    metalMine: buildings.MetalMine,
                    crystalMine: buildings.CrystalMine,
                    deuteriumSynthesizer: buildings.DeuteriumSynthesizer,
                    solarPlant: buildings.SolarPlant,
                    fusionReactor: buildings.FusionReactor,
                    metalStorage: buildings.MetalStorage,
                    crystalStorage: buildings.CrystalStorage,
                    deuteriumStorage: buildings.DeuteriumStorage,
                    updatedAt: new Date(),
                  },
                });
            } catch (error) {
              this.log.warn({ planetId: planet.ID, error: String(error) }, 'Failed to refresh resource buildings');
            }

            try {
              const fac = await this.client.getFacilities(planet.ID);
              await this.db.insert(schema.facilities)
                .values({
                  planetId: planet.ID,
                  roboticsFactory: fac.RoboticsFactory,
                  shipyard: fac.Shipyard,
                  researchLab: fac.ResearchLab,
                  allianceDepot: fac.AllianceDepot,
                  missileSilo: fac.MissileSilo,
                  naniteFactory: fac.NaniteFactory,
                  terraformer: fac.Terraformer,
                  spaceDock: fac.SpaceDock,
                  updatedAt: new Date(),
                })
                .onConflictDoUpdate({
                  target: schema.facilities.planetId,
                  set: {
                    roboticsFactory: fac.RoboticsFactory,
                    shipyard: fac.Shipyard,
                    researchLab: fac.ResearchLab,
                    allianceDepot: fac.AllianceDepot,
                    missileSilo: fac.MissileSilo,
                    naniteFactory: fac.NaniteFactory,
                    terraformer: fac.Terraformer,
                    spaceDock: fac.SpaceDock,
                    updatedAt: new Date(),
                  },
                });
            } catch (error) {
              this.log.warn({ planetId: planet.ID, error: String(error) }, 'Failed to refresh facilities');
            }
          }

          // Persist fleets (full replace — delete all then insert)
          await this.db.delete(schema.fleets);
          for (const fleet of fleets) {
            await this.db.insert(schema.fleets).values({
              id: fleet.ID,
              mission: fleet.Mission,
              returnFlight: fleet.ReturnFlight,
              originGalaxy: fleet.Origin.galaxy,
              originSystem: fleet.Origin.system,
              originPosition: fleet.Origin.position,
              destGalaxy: fleet.Destination.galaxy,
              destSystem: fleet.Destination.system,
              destPosition: fleet.Destination.position,
              arrivalTime: fleet.ArrivalTime,
              metal: fleet.Metal,
              crystal: fleet.Crystal,
              deuterium: fleet.Deuterium,
              updatedAt: new Date(),
            });
          }

          // Persist research
          await this.db.insert(schema.researchTable)
            .values({
              id: 1,
              energyTechnology: research.EnergyTechnology,
              laserTechnology: research.LaserTechnology,
              ionTechnology: research.IonTechnology,
              hyperspaceTechnology: research.HyperspaceTechnology,
              plasmaTechnology: research.PlasmaTechnology,
              combustionDrive: research.CombustionDrive,
              impulseDrive: research.ImpulseDrive,
              hyperspaceDrive: research.HyperspaceDrive,
              espionageTechnology: research.EspionageTechnology,
              computerTechnology: research.ComputerTechnology,
              astrophysics: research.Astrophysics,
              intergalacticResearchNetwork: research.IntergalacticResearchNetwork,
              gravitonTechnology: research.GravitonTechnology,
              weaponTechnology: research.WeaponTechnology,
              shieldingTechnology: research.ShieldingTechnology,
              armourTechnology: research.ArmourTechnology,
              updatedAt: new Date(),
            })
            .onConflictDoUpdate({
              target: schema.researchTable.id,
              set: {
                energyTechnology: research.EnergyTechnology,
                laserTechnology: research.LaserTechnology,
                ionTechnology: research.IonTechnology,
                hyperspaceTechnology: research.HyperspaceTechnology,
                plasmaTechnology: research.PlasmaTechnology,
                combustionDrive: research.CombustionDrive,
                impulseDrive: research.ImpulseDrive,
    ```

    **Main Entry Point** (`packages/bot/src/main.ts`):

    Wire everything together per the startup sequence: load config → init logger → init DB → create rate limiter → create client → login → create game state → start:
    ```typescript
    import { loadConfig } from './config/config-loader.js';
    import { createLogger } from './utils/logger.js';
    import { RateLimiter } from './client/rate-limiter.js';
    import { OgamedClient } from './client/ogamed-client.js';
    import { initDatabase } from './db/index.js';
    import { GameState } from './state/game-state.js';

    async function main() {
      // 1. Load configuration per D-04, D-05, D-07
      const configPath = process.env.CONFIG_PATH || './config.yaml';
      const config = loadConfig(configPath);

      // 2. Initialize logger per D-18
      const log = createLogger('bot', config.logLevel);
      log.info({ configPath }, 'Starting OGame Bot');

      // 3. Initialize database per D-08, D-10
      const dbPath = process.env.DB_PATH || './data/bot.db';
      const migrationsFolder = './drizzle';
      const { db, sqlite } = initDatabase(dbPath, migrationsFolder, log);

      // 4. Create rate limiter and ogamed client per D-14, D-11
      const rateLimiter = new RateLimiter(config.rateLimit, log);
      const client = new OgamedClient(config.ogamed.url, rateLimiter, log);

      // 5. Authenticate with ogamed per INFRA-01
      try {
        log.info('Authenticating with ogamed...');
        await client.login();
        const serverTime = await client.getServerTime();
        log.info({ serverTime }, 'Connected to ogamed successfully');
      } catch (error) {
        log.fatal({ error: String(error) }, 'Failed to connect to ogamed — exiting');
        sqlite.close();
        process.exit(1);
      }

      // 6. Create and start game state manager per D-09, INFRA-02
      const stateRefreshMs = 60000; // 60 seconds base interval
      const gameState = new GameState(db, client, stateRefreshMs, log);
      await gameState.start();

      log.info('Bot started successfully — game state refresh active');

      // 7. Graceful shutdown
      const shutdown = async (signal: string) => {
        log.info({ signal }, 'Shutting down...');
        gameState.stop();
        sqlite.close();
        log.info('Shutdown complete');
        process.exit(0);
      };

      process.on('SIGINT', () => shutdown('SIGINT'));
      process.on('SIGTERM', () => shutdown('SIGTERM'));
    }

    main().catch((error) => {
      console.error('Fatal error:', error);
      process.exit(1);
    });
    ```
