**10. Create `packages/dashboard/src/App.tsx`** — Root component with state management:
```tsx
import { createSignal, onMount, onCleanup } from 'solid-js';
import { fetchAllState, fetchBuildEvents, fetchFleetSaveEvents, fetchFarmAttacks } from './api/client';
import { createWSClient } from './api/websocket';
import type { APIPlanet, APIFleet, APIResearch, APIBuildEvent, APIFleetSaveEvent, APIFarmAttack } from '@ogame-bot/shared';
import Header from './components/Header';
import EmpireOverview from './components/EmpireOverview';
import FleetMovements from './components/FleetMovements';
import ActivityFeed from './components/ActivityFeed';

export default function App() {
  const [planets, setPlanets] = createSignal<APIPlanet[]>([]);
  const [fleets, setFleets] = createSignal<APIFleet[]>([]);
  const [research, setResearch] = createSignal<APIResearch | null>(null);
  const [buildEvents, setBuildEvents] = createSignal<APIBuildEvent[]>([]);
  const [fleetSaveEvents, setFleetSaveEvents] = createSignal<APIFleetSaveEvent[]>([]);
  const [farmAttacks, setFarmAttacks] = createSignal<APIFarmAttack[]>([]);
  const [connected, setConnected] = createSignal(false);
  const [lastUpdate, setLastUpdate] = createSignal<Date | null>(null);

  // Load initial state via REST
  onMount(async () => {
    const [state, builds, saves, attacks] = await Promise.all([
      fetchAllState(),
      fetchBuildEvents(),
      fetchFleetSaveEvents(),
      fetchFarmAttacks(),
    ]);
    setPlanets(state.planets);
    setFleets(state.fleets);
    setResearch(state.research);
    setBuildEvents(builds);
    setFleetSaveEvents(saves);
    setFarmAttacks(attacks);
    setLastUpdate(new Date());

    // Connect WebSocket for real-time updates
    const ws = createWSClient((msg) => {
      switch (msg.type) {
        case 'state_update':
          setPlanets(msg.data.planets);
          setFleets(msg.data.fleets);
          setResearch(msg.data.research);
          break;
        case 'build_event':
          setBuildEvents(prev => [msg.data, ...prev].slice(0, 50));
          break;
        case 'fleet_save_event':
          setFleetSaveEvents(prev => [msg.data, ...prev].slice(0, 20));
          break;
        case 'farm_attack':
          setFarmAttacks(prev => [msg.data, ...prev].slice(0, 50));
          break;
      }
      setLastUpdate(new Date());
    });
    ws.connect();
    onCleanup(() => ws.disconnect());
  });

  return (
    <div class="dashboard">
      <Header connected={connected()} lastUpdate={lastUpdate()} />
      <main>
        <EmpireOverview planets={planets()} />
        <FleetMovements fleets={fleets()} />
        <ActivityFeed buildEvents={buildEvents()} fleetSaveEvents={fleetSaveEvents()} farmAttacks={farmAttacks()} />
      </main>
    </div>
  );
}
```

import type { APIFleet } from '@ogame-bot/shared';
import { For, Show } from 'solid-js';

const missionNames: Record<number, string> = {
  1: 'Attack', 3: 'Transport', 4: 'Deploy', 5: 'Hold',
  6: 'Espionage', 8: 'Harvest', 9: 'Moon Destruction', 15: 'Expedition',

function formatArrival(arrivalTime: number): string {
  const date = new Date(arrivalTime * 1000); // Unix timestamp
  const now = new Date();
  const diff = date.getTime() - now.getTime();
  if (diff <= 0) return 'Arrived';
  const mins = Math.floor(diff / 60000);
  const hrs = Math.floor(mins / 60);
  if (hrs > 0) return `${hrs}h ${mins % 60}m`;
  return `${mins}m`;
}

export default function FleetMovements(props: { fleets: APIFleet[] }) {
  return (

          <tbody>
            <For each={props.fleets}>{fleet => (

**5. Create `packages/dashboard/src/components/ActivityFeed.tsx`:**
Combined activity feed showing build events, fleet-save events, and farm attacks, sorted by time (most recent first).

Show each event type with a colored badge and icon:
- Build events: 🔨 blue badge "BUILD" — "MetalMine 20→21 on planet #1 (cost: 50K metal)"
- Fleet-save events: 🛡️ yellow badge "FLEET SAVE" — "Planet #1 → Planet #2, recalled: true"
- Farm attacks: ⚔️ green badge "FARM" — "Attack → 1:100:5, loot: 50K metal, 20K crystal"

```tsx
import type { APIBuildEvent, APIFleetSaveEvent, APIFarmAttack } from '@ogame-bot/shared';
import { For } from 'solid-js';

// Union type for display
type ActivityItem = 
  | { kind: 'build'; time: string; data: APIBuildEvent }
  | { kind: 'fleet-save'; time: string; data: APIFleetSaveEvent }
  | { kind: 'farm'; time: string; data: APIFarmAttack };

export default function ActivityFeed(props: {
  buildEvents: APIBuildEvent[];
  fleetSaveEvents: APIFleetSaveEvent[];
  farmAttacks: APIFarmAttack[];
}) {
  // Merge and sort all events by time (most recent first)
  const items = (): ActivityItem[] => {
    const all: ActivityItem[] = [
      ...props.buildEvents.map(e => ({ kind: 'build' as const, time: e.createdAt, data: e })),
      ...props.fleetSaveEvents.map(e => ({ kind: 'fleet-save' as const, time: e.sentAt.toString(), data: e })),
      ...props.farmAttacks.map(e => ({ kind: 'farm' as const, time: e.sentAt.toString(), data: e })),
    ];
    all.sort((a, b) => b.time.localeCompare(a.time));
    return all.slice(0, 30); // Show last 30 events
  };

  return (
    <section class="activity-feed">
      <h2>Activity Feed</h2>
      <div class="feed-list">
        <For each={items()}>{item => (
          <div class={`feed-item ${item.kind}`}>
            {/* render based on item.kind with appropriate badge and text */}
          </div>
        )}</For>
      </div>
    </section>
  );
}
```
